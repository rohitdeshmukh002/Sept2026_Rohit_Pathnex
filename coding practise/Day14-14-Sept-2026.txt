# Day 14 

## Jenkins Pipeline — Scheduled Build

pipeline {
    agent any
    environment {
        INSTITUTE_NAME = "Rohit"
    }
    triggers {
        cron('H 5 * * *') // every day at 5 AM
    }
    stages {
        stage('Checkout') {
            steps {
                git url: 'https://github.com/rohitdeshmukh002/employee-management-system.git'
            }
        }
        stage('Build') {
            steps {
                sh 'mvn clean package'
            }
        }
    }
}
