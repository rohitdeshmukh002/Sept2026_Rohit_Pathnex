#1. ANSIBLE


- name: Create pathnex-admin user
  hosts: all
  become: true

  tasks:
    - name: Create user
      ansible.builtin.user:
        name: pathnex-admin
        shell: /bin/bash
        create_home: true
        state: present

    - name: Add SSH key
      ansible.posix.authorized_key:
        user: pathnex-admin
        key: "{{ lookup('file', '~/.ssh/id_rsa.pub') }}"
        state: present

#!/bin/bash

echo "1) IP"
echo "2) Disk"
echo "3) Memory"

read -p "Enter choice: " choice

case $choice in
  1)
    hostname -I
    ;;
  2)
    df -h
    ;;
  3)
    free -h
    ;;
  *)
    echo "Invalid choice"
    ;;
esac

#Automate all tasks using a Declarative Jenkins Pipeline.

Stages:

1. Infrastructure
   - Terraform EC2
   - Project = "Pathnex-Training"

2. App Deployment
   - Ansible user creation
   - SSH setup
   - Kubernetes deployment

3. Monitoring Checks
   - IP
   - Disk
   - Memory

Environment variables:

INSTITUTE_NAME = "Pathnex"
TEAM           = "DevOps"
ENV            = "prod"
PROJECT        = "Pathnex-Training"


Example Jenkinsfile:

pipeline {

    agent any

    environment {
        INSTITUTE_NAME = "Pathnex"
        TEAM           = "DevOps"
        ENV            = "prod"
        PROJECT        = "Pathnex-Training"
    }

    stages {

        stage('Infrastructure') {
            steps {
                sh '''
                    terraform init
                    terraform apply -auto-approve
                '''
            }
        }

        stage('App Deployment') {
            steps {
                sh '''
                    ansible-playbook ansible.yml
                    kubectl apply -f deployment.yml
                    kubectl apply -f service.yml
                '''
            }
        }

        stage('Monitoring Checks') {
            steps {
                sh '''
                    echo "IP:"
                    hostname -I

                    echo "Disk:"
                    df -h

                    echo "Memory:"
                    free -h
                '''
            }
        }
    }
}
