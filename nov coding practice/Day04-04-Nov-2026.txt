## Terraform — EC2 with Key Pair (c6i.8xlarge)

provider "aws" {
    region = "ap-south-1"
}

provider "aws_key_pair" "PathnexKey" {
    key_name    = "PathnexKey"
    public_key  = "ssh-rsa AAAA ...."

    tags = {
        Name = "pathnex-key"
    }
}

resource "aws_instance" "PathnexEC2" {
    ami             = "ami-dsds3432sdfds"
    instance_type   = "c6i.8xlarge"
    key_name        = aws_key_pair.PathnexKey.key_name

    tags    = {
        Name = "Pathnex-EC2"
    }
}

# Parallel Builds
## Jenkins Pipeline — Parallel Build & Test
You will learn how to **run build and test in parallel**.

pipeline {
    agent any
    stages {
        stage ('Parallel tasks') {
            parallel {
                stage('Build') {
                    steps {
                        sh 'mvn clean compile'
                    }
                }
                stage ('Test') {
                    steps {
                        sh 'mvn test'
                    }
                }
            }
        }
    }
}

## GitLab CI — Parallel Jobs
You will learn how to **run parallel jobs in GitLab CI**.

stages:
    - build-test

maven-build:
    stage: build-test
    image: maven:3.7.1-jdk-17
    script:
        - git clone https://github.com/Pathnex/sample-java-app.git
        - cd sample-java-app
        - mvn clean compile
    parallel: 1

maven-test:
    stage: build-test
    image: maven:3.8.1-jdk-17
    script:
        - cd sample-java-app
        - mvn test

## Ansible — Install and Configure Nginx

- name: install and configure nginx
  hosts: all
  become: yes

  tasks:
    - name: install nginx
      yum:
        name: nginx
        state: present
    - name: configure nginx
      template:
        scr: nginx.conf.j2
        dest: /etc/nginx/nginx.conf
    - name: start nginx service
      service:
        name: nginx
        state: started
        enabled: yes

## Terraform — EC2 with Security Group

resource "aws_security_group" "my_Sg" {
    name        = "server_sg"
    description = "allow ssh and https"

    ingress {
        from_port   = 22
        to_port     = 22
        protocol    = "tcp"
        cidr_blocks = ["0.0.0.0/0"]
    }

    ingress {
        from_port   = 80
        to_port     = 80
        protocol    = "tcp"
        cidr_blocks = ["0.0.0.0/0"]
    }

    egress  {
        from_port   = 0
        to_port     = 0
        protocol    = "tcp"
        cidr_blocks = ["0.0.0.0/0"]
    }
}
