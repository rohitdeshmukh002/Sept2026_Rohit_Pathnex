# Caching Dependencies

## 🔹 Jenkins Pipeline — Cache Maven Dependencies
You will learn how to **reuse dependencies to speed up builds**.

pipeline {
    agent any
    environment {
        INSTITUTE_NAME = "Pathnex"
    }
    stages {
        stage ('Checkout') {
            steps {
            git url: 'https://github.com/rohitdeshmukh002.cohert.git'
            }
        }
        stage ('Build') {
            steps {
                sh '''
                if [-d ".m2"]; then
                    echo "using cached dependencies"
                else 
                    mvn clean compile
                fi
                '''
            }
        }
    }
}

## 🔹 Ansible — Create Docker Containers for Multiple Applications

```yaml

- name: steup docker container for Multiple apps
  hosts: all
  become: yes

  tasks:
    - name: pull web app image
      docker_image:
        name: my-web-app
        source: pull
    
    - name: pull database app image
      docker_image:
        name: my-database-app
        source: pull
    
    - name: run web app container
      docker_container:
        name: web-app
        image: my-web-app
        state: started
        published_ports:
            - "8080:80"
    
    - name: Run Database App Container
      docker_container:
        name: database-app
        image: my-database-app
        state: started
        published_ports:
          - "3306:3306"

🔹 Terraform — VPC with Private and Public Subnet

resource "aws_vpc" "main" {
  cidr_block = "10.0.0.0/16"
}

resource "aws_subnet" "public" {
  vpc_id                  = aws_vpc.main.id
  cidr_block              = "10.0.1.0/24"
  availability_zone       = "us-east-1a"
  map_public_ip_on_launch = true
}

resource "aws_subnet" "private" {
  vpc_id            = aws_vpc.main.id
  cidr_block        = "10.0.2.0/24"

🔹 Kubernetes — Deploying StatefulSet

apiVersion: apps/v1
kind: StatefulSet
metadata:
  name: pathnex-db
spec:
  serviceName: "pathnex-db"
  replicas: 3
  selector:
    matchLabels:
      app: pathnex-db
  template:
    metadata:
      labels:
        app: pathnex-db
    spec:
      containers:
        - name: db
          image: mysql:5.7
          volumeMounts:
            - mountPath: /var/lib/mysql
              name: mysql-storage
  volumeClaimTemplates:
    - metadata:
        name: mysql-storage
      spec:
        accessModes: ["ReadWriteOnce"]
        resources:
          requests:
            storage: 1Gi

🔹 Jenkinsfile — Build and Deploy to Kubernetes

pipeline {
    agent any
    environment {
        IMAGE_NAME = 'pathnex-web-app'
    }
    stages {
        stage ('Build docker image') {
            steps {
                script {
                    docker.build("${env.IMAGE_NAME}")
                }
            }
        }
        stage ('Push dcoker image') {
            steps {
                script {
                    docker.withRegistry('https://docker.io', 'docker-credentials') {
                        dcoker.image("${env.IMAGE_NAME}").push('latest')
                    }
                }
            }
        }
        stage ('Deploy to kubernetes') {
            steps {
                script {
                    sh 'kubectl apply -f deployment.yaml'
                }
            }
        }
    }
}
