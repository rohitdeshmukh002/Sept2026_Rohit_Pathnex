# Parallel Builds
## Jenkins Pipeline — Parallel Build & Test

pipeline {
    agent any
    stages {
        stage('Parallel Tasks') {
            parallel {
                steps('Build') {
                    steps {
                        sh 'nvm clean compile'
                    }
                }
                stag('Test') {
                    steps {
                        sh 'nvm test'
                    }
                }
            }
        }
    }
}
## GitLab CI - parallel jobs

stages:
    - build-test

maven-build:
    stage: build-test
    image: maven: 3.8.1-jdk-17
    script:
        - git clone https://github.com/rohitdeshmukh002/cohert/git
        - cd cohert
        - mvn clean compile
    parallel: 1

maven-test:
    stage: build-test
    image: maven:3.8.1-jdk-17
    script:
        - cd cohert
        - maven test
    parallel: 1


## Ansible - Install apache and start service

---
 - name: Ansible - install apache and start service
   hosts: all
   become: yes
   tasks:
    - name: install apache
      yum:
        name: httpd
        state: present
    - name: start service
      service:
        name: httpd
        state: start
        enabled: yes

## Kubernetes -  Create a replicaset with 3 replicas

appVersion: app/v1
kind: replicaset
metadata:
    name: server-replicaset
spec:
    replicas: 3
    selector:
        matchLabels: pathnex-app
    template:
        metadata:
        labels:
            app: server-app
        spec:
            containers:
                - name: app
                  image: nginx
                  ports:
                    - containerPort: 80
