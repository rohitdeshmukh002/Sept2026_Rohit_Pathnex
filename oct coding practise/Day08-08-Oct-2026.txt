# Day 08 — Variables, VPC, Probes, Loops

## Ansible — Install Packages Using Variables

---
  - name: Install packages for Rohit using variables
    hosts: yes
    become: yes

    vars:
      rohit_packages:
        - tree
        - unzip
        - vim

    tasks:
      - name: Install required packages
        yum:
          name: "{{ rohit_packages }}"
          state: present

## Terraform - Create vpc + Subnet

  region = "ap-south-1"
}

resource "aws_vpc" "server-vpc" {
  cidr_block = "10.0.0.0/16"

## Kubernetes — Add Liveness & Readiness Probes

apiVersion: v1
kind: Pod
metadata:
  name: pathnex-probe-pod
spec:
  containers:
    - name: web
      image: nginx
      livenessProbe:
        httpGet:
          path: /
          port: 80
        initialDelaySeconds: 5
      readinessProbe:
        httpGet:
          path: /
          port: 80
        initialDelaySeconds: 3

## Conditional Stages - Jenkins pipeline

pipeline {
  agent any
  parameters {
        string(name: 'ENVIRONMENT', defaultValue: 'dev', description: 'Deployment Environment')
    }
    environment {
        INSTITUTE_NAME = "Pathnex"
    }
    stages {
      stage('checkout') {
        steps {
            git clone: 'main', url: 'https://github.com/Pathnex/cohert.git'
        }
      }
      state('build') {
        when {
            branch 'main'
        }
        steps {
          sh 'nvm clean compile'
        }
      }
      stage('Deploy') {
        when {
          expression {params.ENVIRONMENT == 'dev'}
        }
        steps {
          sh 'echo Deploying to $Environment by $INSTITUTE_NAME'
        }
      }
    }
}

## Terraform — EC2 with Security Group and EBS Volume

provider "aws" {
  region = "ap-south-1"
}

resource "aws_security_group" "sg-server" {
  name        = "allow_ssh_http"
  description = "allow ssh and http"

  ingress = {
    from_port   = "22"
    to_port     = "22"
    cidr_block  = [0.0.0.0/16]
    protocol    = "tcp"
  }

  ingress = {
    from_port       = "80"
    to_port         = "80"
    cidr_blocks     = ["0.0.0.0/0"]
    protocol        = "tcp"
  }
}

resource "aws_instance" "aws-ec2" {
  ami             = "ami-55454sdsdsdsd45455"
  instance_type   = "t3.micro"
  security_groups = [aws_security_group.allow_ssh_http]

  ebs_block_device  ={
    device_name = "/dev/sdh"
    volume_size = 30
  }
}

## Jenkinsfile — Docker Build and Push

pipeline {
  agent any
  Stages {
    stage ('build docker images') {
      steps {
      script {
        docker.build('abhi-app')
      }
      }
    }
    stage('Push Docker Image') {
            steps {
                script {
                    docker.withRegistry('https://docker.io', 'docker-credentials') {
                        docker.image('pathnex-app').push('latest')
                    }
                }
            }
        }
  }
}
