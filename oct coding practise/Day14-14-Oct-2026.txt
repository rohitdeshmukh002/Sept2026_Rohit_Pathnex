##Terraform - create EC2


provider "aws" {
  region = "ap-south-1"
}

variable "instance_type" {
  description = "EC2 instance type"
  type        = string
  default     = "t3.micro"
}

variable "key_name" {
  description = "Existing AWS EC2 key pair"
  type        = string
  default     = "My_key"
}

# Fetch a current Ubuntu 24.04 AMI for Mumbai
data "aws_ssm_parameter" "ubuntu_ami" {
  name = "/aws/service/canonical/ubuntu/server/24.04/stable/current/amd64/hvm/ebs-gp3/ami-id"
}

resource "aws_vpc" "main" {
  cidr_block           = "10.10.0.0/16"
  enable_dns_support   = true
  enable_dns_hostnames = true

  tags = {
    Name = "devops-vpc"
  }
}

resource "aws_subnet" "public" {
  vpc_id                  = aws_vpc.main.id
  cidr_block              = "10.10.1.0/24"
  map_public_ip_on_launch = true

  tags = {
    Name = "public-subnet"
  }
}

resource "aws_internet_gateway" "igw" {
  vpc_id = aws_vpc.main.id

  tags = {
    Name = "devops-igw"
  }
}

resource "aws_route_table" "public" {
  vpc_id = aws_vpc.main.id

  route {
    cidr_block = "0.0.0.0/0"
    gateway_id = aws_internet_gateway.igw.id
  }

  tags = {
    Name = "public-route-table"
  }
}

resource "aws_route_table_association" "public" {
  subnet_id      = aws_subnet.public.id
  route_table_id = aws_route_table.public.id
}

resource "aws_security_group" "web" {
  name        = "devops-web-sg"
  description = "SSH and HTTP access"
  vpc_id      = aws_vpc.main.id

  ingress {
    description = "SSH - restrict to your public IP"
    from_port   = 22
    to_port     = 22
    protocol    = "tcp"
    cidr_blocks = ["YOUR_PUBLIC_IP/32"]
  }

  ingress {
    description = "HTTP"
    from_port   = 80
    to_port     = 80
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  egress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }

  tags = {
    Name = "devops-web-sg"
  }
}

resource "aws_instance" "my_ec2" {
  ami                         = data.aws_ssm_parameter.ubuntu_ami.value
  instance_type               = var.instance_type
  key_name                    = var.key_name
  subnet_id                   = aws_subnet.public.id
  vpc_security_group_ids      = [aws_security_group.web.id]
  associate_public_ip_address = true

  tags = {
    Name = "ec2-server"
  }
}

output "public_ip" {
  value = aws_instance.my_ec2.public_ip
}

## 2️⃣ Ansible — Install Nginx & Deploy HTML File

---
- name: Install Nginx and deploy HTML file
  hosts: web
  become: true

  tasks:
    - name: Install Nginx
      ansible.builtin.apt:
        name: nginx
        state: present
        update_cache: true

    - name: Start and enable Nginx
      ansible.builtin.service:
        name: nginx
        state: started
        enabled: true

    - name: Deploy HTML page
      ansible.builtin.copy:
        src: index.html
        dest: /var/www/html/index.html
        owner: root
        group: root
        mode: "0644"


# Scheduled Pipelines

## Jenkins Pipeline — Scheduled Build
You will learn how to **schedule a Jenkins pipeline using cron**.

pipeline {
    agent any
    environment {
        INSTITUTE_NAME = "devops"
    }
    triggers {
        cron('H 2 * * *')
    }
    stages {
        stage ('checkout') {
            steps {
                git url: 'https://github.com/rohitdeshmukh002/cohert.git'
            }
        }
        stage ('build') {
            steps {
                sh 'mvn clean package'
            }
        }
    }
}

## GitLab CI — Scheduled Pipeline
You will learn how to **schedule GitLab CI pipelines via cron**.

stages:
    - build
    - test

build:
    stage: build
    image: maven:3.8.7-jdk-17

        - git clone https://github.com/rohitdeshmukh002/cohert.git
        - cd cohert
        - mvn clean package

test:
    stage: test
    image: maven:3.8.7-jdk-17

        - cd cohert
        - maven test
