## Ansible - Install apache and start service

---
  name: install apache and start service
  hosts: any
  become: yes

  tasks:
    - name: install apache
      yum:
        name: httpd
        state: present
    
    - name: start apache
      service:
        name: httpd
        state: started
        enabled: yes

## Terraform EC2 creation with Security-group

  region = "ap-south-1"
}

resource "aws_instance" "server-ec2"{
  ami = "ami-ocsjb23323dsddsf"
  instance_type = "t2.micro"
  key_pair = "key_pair"
  vpc_security_group_ids = ["aws_security_group.private-sg.id"]

  tags = {
    Name = "rohit"
  }
}

resource "aws_security_group" "private-sg" {
  Name: "private-sg"
  description : private Security group

  ingress {
    to_port= 22
    from_port= 22
    protocol= "tcp"
    cidr_blocks = ['0.0.0.0/0']
  }

  egress {
    to_port     = 0
    from_port   = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }
}

## Kubernates -  create a replicaset with 3 replicas

  name: rogit-replicaset

  replicas: 3 
  selector:
    matchLabels:
      app: rohit-app  
    template:
      metadata:
      labels:
        app: rohit-app
    spec:
      containers:
        - name: app 
          image: nginx
