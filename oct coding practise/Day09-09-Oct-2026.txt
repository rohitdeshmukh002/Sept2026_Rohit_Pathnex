## Ansible — Setup Docker Container for Nginx

- name: Setup Docker containers for Nginx
  hosts: all
  become: yes
  tasks:
    - name: pull nginx image
      docker_image:
        name: nginx
        source: pull
    - name: run nginx container
      docker_container:
        name: nginx-container
        image: nginx
        state: started
        published_ports:
            "8080:80"
