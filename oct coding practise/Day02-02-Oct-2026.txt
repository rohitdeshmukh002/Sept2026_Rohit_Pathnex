# Ansible

- name: Install Nginx inside the server and enable
  hosts: all
  become: yes

  tasks:
 - name: Install nginx
      yum:
        name: nginx
        state: present

    - name: Enable nginx
      service:
        name: nginx
        state: started
        enabled: yes

# Kubernetes - Deployment with 2 Replicas
appVersion: apps/v1
kind: Deployment
metadata:
    name: Google-deployment
spec:
    replicas: 2
    selector:
        matchLabels:
            app: google-app
    template:
        metadata:
        labels:
            app: google-app
    spec:
        containers:
        - name: app
          images: nginx
          ports:
            - containerPort: 80

# Ansible docker install

- name: Install Docker and start service
  hosts: all
  become: yes

  tasks:
    - name: Install Docker
      apt:
        name: docker.io
        state: present
        update_cache: yes
      tags:
        - install

    - name: Start Docker
      service:
        name: docker
        state: started
      tags:
        - start
